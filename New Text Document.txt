        // Custom Print Calculation
        function calculateCustomPrint() {
            const pages = parseInt(document.getElementById('calcPages').value) || 0;
            const type = document.getElementById('calcType').value;
            const binding = document.getElementById('calcBinding').value;

            let pageRate = (type === 'bw') ? 0.50 : 1.50;
            let bindingCost = (binding === 'wire') ? 15 : (binding === 'hard') ? 30 : 5;

            let total = (pages * pageRate) + bindingCost;
            document.getElementById('calcTotalPrice').textContent = `${total.toFixed(0)} ج.م`;
            return total;
        }

        // Add Custom Print Job to Cart
        function addCustomPrintToCart() {
            const title = document.getElementById('calcTitle').value.trim() || 'مذكرة مخصصة للطباعة';
            const pages = parseInt(document.getElementById('calcPages').value) || 1;
            const typeText = (document.getElementById('calcType').value === 'bw') ? 'أبيض وأسود' : 'ألوان';
            const price = calculateCustomPrint();

            const customItem = {
                id: 'custom_' + Date.now(),
                title: `طباعة: ${title} (${pages} صفحة - ${typeText})`,
                price: price,
                qty: 1,
                isCustom: true
            };

            cart.push(customItem);
            saveCart();
            showToast('تمت إضافة طلب الطباعة إلى السلة بنجاح');
            openCartModal();
        }

        // Add Regular Product to Cart
        function addToCart(prodId) {
            const item = products.find(p => p.id === prodId);
            if (!item) return;

            const existing = cart.find(c => c.id === prodId);
            if (existing) {
                existing.qty += 1;
            } else {
                cart.push({ ...item, qty: 1 });
            }

            saveCart();
            showToast(`تمت إضافة "${item.title}" للسلة`);
        }

        // Update Quantity in Cart
        function updateCartQty(id, delta) {
            const item = cart.find(c => c.id === id);
            if (!item) return;

            item.qty += delta;
            if (item.qty <= 0) {
                cart = cart.filter(c => c.id !== id);
            }

            saveCart();
        }

        // Save Cart & Update Interface
        function saveCart() {
            localStorage.setItem('future_bookstore_cart', JSON.stringify(cart));
            updateCartUI();
        }

        // UI Refresh for Cart
        function updateCartUI() {
            const count = cart.reduce((sum, item) => sum + item.qty, 0);
            document.getElementById('cartBadge').textContent = count;

            const container = document.getElementById('cartItemsContainer');
            if (cart.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-12 text-gray-400">
                        <i class="fa-solid fa-cart-flatbed text-4xl mb-3"></i>
                        <p class="text-xs font-bold">سلة التسوق فارغة حالياً</p>
                    </div>
                `;
                document.getElementById('cartGrandTotal').textContent = `0 ج.م`;
                return;
            }

            let grandTotal = 0;
            container.innerHTML = cart.map(item => {
                let itemTotal = item.price * item.qty;
                grandTotal += itemTotal;
                return `
                    <div class="flex items-center justify-between bg-gray-50 dark:bg-gray-800/60 p-3.5 rounded-2xl border border-gray-100 dark:border-gray-800">
                        <div class="flex-1 pl-3">
                            <h5 class="text-xs font-bold text-gray-900 dark:text-white line-clamp-1">${item.title}</h5>
                            <span class="text-xs text-amber-600 dark:text-amber-400 font-extrabold">${item.price} ج.م</span>
                        </div>
                        <div class="flex items-center gap-2">
                            <button onclick="updateCartQty('${item.id}', -1)" class="w-7 h-7 bg-white dark:bg-gray-700 text-gray-700 dark:text-gray-200 rounded-lg font-black border border-gray-200 dark:border-gray-600 flex items-center justify-center">-</button>
                            <span class="text-xs font-extrabold w-4 text-center">${item.qty}</span>

                        </div>
                    </div>
                `;
            }).join('');

            document.getElementById('cartGrandTotal').textContent = `${grandTotal} ج.م`;
        }
